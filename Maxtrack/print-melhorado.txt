    // Botão de Print (capturar frame) - VERSÃO MELHORADA
    // Captura diretamente do elemento de vídeo usando canvas (mais rápido e confiável)
    const printBtn = document.createElement('button');
    printBtn.className = 'maxtrack-video-print-btn';
    printBtn.innerHTML = '📷 Print';
    printBtn.title = 'Capturar frame do vídeo (P)';
    printBtn.addEventListener('click', async () => {
      try {
        console.log('📸 Iniciando captura do frame do vídeo...');
        
        // Pausar o vídeo para captura
        const wasPlaying = !videoClone.paused;
        if (wasPlaying) {
          videoClone.pause();
        }
        
        // Criar canvas para capturar o frame do vídeo
        const canvas = document.createElement('canvas');
        canvas.width = videoClone.videoWidth || videoClone.clientWidth;
        canvas.height = videoClone.videoHeight || videoClone.clientHeight;
        
        const ctx = canvas.getContext('2d');
        
        // Desenhar o frame atual do vídeo no canvas
        ctx.drawImage(videoClone, 0, 0, canvas.width, canvas.height);
        
        console.log('📸 Frame capturado, gerando arquivo...');
        
        // Converter para blob e fazer download
        canvas.toBlob((blob) => {
          if (blob) {
            const url = URL.createObjectURL(blob);
            const a = document.createElement('a');
            // Gerar nome do arquivo com timestamp
            const timestamp = new Date().toISOString().replace(/[:.]/g, '-').slice(0, -5);
            a.href = url;
            a.download = `maxtrack-frame-${timestamp}.png`;
            document.body.appendChild(a);
            a.click();
            document.body.removeChild(a);
            URL.revokeObjectURL(url);
            
            showNotification('📸 Frame capturado!', 'success', 'Imagem salva nos downloads');
            console.log('✅ Frame capturado com sucesso');
          } else {
            throw new Error('Não foi possível gerar o blob da imagem');
          }
        }, 'image/png');
        
        // Retomar reprodução se estava tocando
        if (wasPlaying) {
          setTimeout(() => videoClone.play(), 200);
        }
        
      } catch (erro) {
        console.error('❌ Erro ao capturar frame:', erro);
        showNotification('❌ Erro ao capturar', 'error', 'Tente novamente');
      }
    });
