// ============================================================
// BACKGROUND SCRIPT - Creare Extension
// Gerencia atualizações automáticas em segundo plano
// ============================================================

const GITHUB_CONFIG = {
  // CONFIGURE AQUI:
  owner: 'expressodouglasguilherme-hub',  // Seu username
  repo: 'extensoes',                       // Nome do repositório
  branch: 'main',                          // Branch (main ou master)
  folder: 'Creare',                        // Pasta da extensão
  checkInterval: 15 * 60 * 1000,          // Verifica a cada 15 minutos (em milissegundos)
  
  get versionUrl() {
    return `https://raw.githubusercontent.com/${this.owner}/${this.repo}/${this.branch}/${this.folder}/version.json`;
  }
};

// Verifica atualizações
async function checkForUpdates(manual = false) {
  try {
    const currentVersion = chrome.runtime.getManifest().version;
    
    console.log('🔍 Verificando atualizações...');
    console.log('📊 Versão atual:', currentVersion);

    const response = await fetch(GITHUB_CONFIG.versionUrl + '?t=' + Date.now(), {
      cache: 'no-cache'
    });

    if (!response.ok) {
      throw new Error(`HTTP ${response.status}`);
    }

    const versionData = await response.json();
    console.log('📊 Versão no GitHub:', versionData.version);

    if (isNewerVersion(versionData.version, currentVersion)) {
      console.log('🎉 Nova versão disponível!');
      
      // Mostra notificação
      chrome.notifications.create({
        type: 'basic',
        iconUrl: 'icon128.png',
        title: `Creare v${versionData.version} disponível!`,
        message: versionData.changelog || 'Nova atualização disponível. Recarregue a extensão.',
        priority: 2,
        buttons: [
          { title: 'Atualizar agora' }
        ]
      });

      // Salva info da atualização
      await chrome.storage.local.set({
        updateAvailable: {
          version: versionData.version,
          changelog: versionData.changelog,
          checkDate: Date.now()
        }
      });

    } else {
      console.log('✅ Extensão está atualizada!');
      if (manual) {
        chrome.notifications.create({
          type: 'basic',
          iconUrl: 'icon128.png',
          title: 'Creare está atualizado!',
          message: `Você tem a versão mais recente (v${currentVersion})`,
          priority: 1
        });
      }
    }

    // Salva última verificação
    await chrome.storage.local.set({
      lastUpdateCheck: Date.now()
    });

  } catch (error) {
    console.error('❌ Erro ao verificar atualizações:', error);
  }
}

// Compara versões
function isNewerVersion(remoteVersion, currentVersion) {
  const remote = remoteVersion.split('.').map(Number);
  const current = currentVersion.split('.').map(Number);

  for (let i = 0; i < 3; i++) {
    if (remote[i] > current[i]) return true;
    if (remote[i] < current[i]) return false;
  }
  return false;
}

// Quando a extensão é instalada/atualizada
chrome.runtime.onInstalled.addListener((details) => {
  if (details.reason === 'install') {
    console.log('🎉 Creare instalado com sucesso!');
    // Verifica imediatamente após instalar
    setTimeout(() => checkForUpdates(), 3000);
  } else if (details.reason === 'update') {
    const version = chrome.runtime.getManifest().version;
    console.log(`✅ Creare atualizado para v${version}`);
    
    chrome.notifications.create({
      type: 'basic',
      iconUrl: 'icon128.png',
      title: `Creare atualizado! v${version}`,
      message: 'Todas as melhorias foram aplicadas com sucesso.',
      priority: 1
    });
    
    // Verifica novamente após 1 minuto
    setTimeout(() => checkForUpdates(), 60000);
  }
});

// Quando o navegador inicia
chrome.runtime.onStartup.addListener(() => {
  console.log('🚀 Creare iniciado com o navegador');
  // Verifica atualizações 10 segundos após o navegador iniciar
  setTimeout(() => checkForUpdates(), 10000);
});

// Agenda verificações periódicas
chrome.alarms.create('checkUpdates', {
  periodInMinutes: GITHUB_CONFIG.checkInterval / (60 * 1000)
});

chrome.alarms.onAlarm.addListener((alarm) => {
  if (alarm.name === 'checkUpdates') {
    checkForUpdates();
  }
});

// Listener para clique na notificação
chrome.notifications.onButtonClicked.addListener((notificationId, buttonIndex) => {
  if (buttonIndex === 0) {
    // Botão "Atualizar agora" clicado
    chrome.runtime.reload();
  }
});

// Listener para mensagens do content script
chrome.runtime.onMessage.addListener((request, sender, sendResponse) => {
  if (request.action === 'checkUpdates') {
    checkForUpdates(true);
    sendResponse({ status: 'checking' });
  }
  return true;
});

console.log('🔄 Background script carregado! Auto-update ativo.');
console.log(`⏰ Verificações a cada ${GITHUB_CONFIG.checkInterval / (60 * 1000)} minutos`);
console.log('📋 Também verifica: ao iniciar navegador, ao instalar/atualizar extensão');
